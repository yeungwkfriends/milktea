const BACKEND_URL = "https://milkteaback.vercel.app/api/chat"; async function sendMessage() { const text = input.value.trim(); if (!text) return; appendMessage('user', text); input.value = ""; setLoading(true); try { const response = await fetch(BACKEND_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: text }) // Make sure this is "message" }); const data = await response.json(); if (data.text) { appendMessage('bot', data.text); } else { throw new Error(data.error || "No response"); } } catch (e) { console.error("Connection Error:", e); appendMessage('bot', "Connection failed. Please check the Vercel URL."); } finally { setLoading(false); } }
